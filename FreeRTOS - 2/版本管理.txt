void motor_up(void)
{
	//舵机 上升
	TIM4->CCR1=INITIAL1; 
	TIM4->CCR2=INITIAL1;
	//步进电机  张开
	HAL_GPIO_WritePin(GPIOC,GPIO_PIN_0,GPIO_PIN_SET); //向上
	HAL_GPIO_WritePin(GPIOC,GPIO_PIN_2,GPIO_PIN_SET);	//使能《》
	HAL_Delay(1500);
	HAL_GPIO_TogglePin(GPIOC,GPIO_PIN_2);				//不能
}
void motor_down(void)
{
	//步进电机  收缩
	HAL_GPIO_WritePin(GPIOC,GPIO_PIN_0,GPIO_PIN_RESET); //向上
	HAL_GPIO_WritePin(GPIOC,GPIO_PIN_2,GPIO_PIN_SET);	//使能
	HAL_Delay(1500);
	HAL_GPIO_TogglePin(GPIOC,GPIO_PIN_2);				//不能
	//舵机 下降
	TIM4->CCR1=MOTOR_DOWN1; 
	TIM4->CCR2=MOTOR_DOWN2;
	HAL_Delay(800);
}
